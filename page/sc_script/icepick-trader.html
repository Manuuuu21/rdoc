      Icepick Trader

      An Ice Pick trader opens costume garment, aura, and headgear shops that accept Ice Picks as currency.

      • Save the downloaded file as npc/custom/manuel_npc/icepick_trader.txt.
      • Add npc: npc/custom/manuel_npc/icepick_trader.txt to npc/scripts_custom.conf if it is not already loaded.

      //===== By:==================================================
//= Manuel Sintos
// @reloadnpcfile npc/custom/manuel_npc/icepick_trader.txt

prontera,141,170,5	script	Ice Pick Trader	564,{

	switch(select("Costume Garments:Costume Aura:Costume Headgear")) {
		case 1:
			callshop "C_Garments",0;
			end;
		break;
		case 2:
			callshop "C_Aura",0;
			end;
		break;
		case 3:
			callshop "C_HG",0;
			end;
		break;
	}
	/*
	OnInit:
		waitingroom "Ice Pick Trader",0;
	end;
	*/
}
