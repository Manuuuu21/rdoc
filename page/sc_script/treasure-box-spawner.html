      Treasure Box Spawner

      A treasure-box NPC spawns a box at a random location and calls the shared reward function when the box is destroyed.

      Download tbox_spawner.txt
      Source filetbox_spawner.txt
      Script length54 lines
      NPC locationsTreasure Box Spawner — prontera (214, 219)

      Setup notes

      • Save the downloaded file as npc/custom/manuel_npc/tbox_spawner.txt.
      • Add npc: npc/custom/manuel_npc/tbox_spawner.txt to npc/scripts_custom.conf if it is not already loaded.
      • Review the item IDs, maps, sprites, prices, and access levels used by this script for your server.
      • Load the external functions called by this script: eventRewards.
      • The eventRewards function is defined in eventrewards.txt in the original source folder. It is a separate dependency, outside this showcase download.

      Script source

      Original script from manuel_npc/tbox_spawner.txt. Author credits and source comments are preserved.

      tbox_spawner.txt
      //===== By:==================================================
// @reloadnpcfile npc/custom/manuel_npc/tbox_spawner.txt

prontera,214,219,3	script	Treasure Box Spawner	735,{
	mes "[ Treasure Box Spawner ]";
	mes "Hi there! I'm the Treasure Box Spawner.";
	mes "Every hour, after you destroyed the box, a treasure box will appear in random locations.";
	mes "Good luck in finding them!";
	close;

	OnInit:
		// Spawn the first treasure box when the server starts
		goto OnSpawnTreasureBox;
	end;
	
	OnTBoxDestroyed:
		// Check if a player is attached (killed by player)
		if (playerattached()) {
			// Announce player destroyed the Tbox
			announce "The Treasure Box has been found and destroyed by [ " + strcharinfo(0) + "! ]", 0;
	
			// Reward the player who killed the treasure box
			callfunc("eventRewards");
		} 
		else if ($killermobgid > 0) {
			announce "The Treasure Box was destroyed by [ " + getunitname($killermobgid) + "! ]", 0;
			$killermobgid = 0; // Reset variable
		}

		// Start a 1-hour cooldown before spawning the next box
		stopnpctimer;
		setnpctimer 0;
		startnpctimer;
	end;
	
	OnTimer3600000:
		// 1 hour (3,600,000 ms) has passed, stop the timer and spawn the box
		stopnpctimer;
		goto OnSpawnTreasureBox;
	end;

OnSpawnTreasureBox:
	// Variables to hold random coordinates
	.@map$ = "prontera";
	// Find a random empty cell in Prontera and store it within .@x and .@y
	getfreecell(.@map$, .@x, .@y); 

	// Spawn the treasure box at the random walkable location
	monster .@map$, .@x, .@y, "Treasure Box", 2288, 1, "Treasure Box Spawner::OnTBoxDestroyed";

	// Announce that the treasure box has respawned
	announce "A new Treasure Box has appeared in Prontera!", 0;
	end;
}
