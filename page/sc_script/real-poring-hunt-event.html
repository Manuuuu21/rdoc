      Real Poring Hunt Event

      An automated hunt hides a real Poring among decoys, provides a warper, and rewards the player who defeats the correct monster.

      //===== rAthena Script =======================================
//= Poring Hunt Event
//===== By: rAthena AI Assistant ============================
//= This script creates an automatic Poring hunting event
//= every 1.5 hours. A "real" Poring is hidden among fake ones.
//= The first player to kill the real Poring wins a reward.
//= @reloadnpcfile npc/custom/manuel_npc/poringkill_event.txt
//============================================================
-	script	PoringEvent	-1,{ // Floating NPC for global event logic and timers (no sprite).

OnInit:
	.num_fakePlayers = 2; // base on array data of fake players
	
	// High Wizard
	setarray .mob_names$, "DonutAce", "MinorityZ";
	setarray .job_id[0], 4010, 4010;
	setarray .gender[0], 1, 1;
	setarray .hairstyle[0], rand(0,42), rand(0,42);
	setarray .haircolor[0], rand(0,8), rand(0,8);
	setarray .weapon_id[0], 1644, 1644;
	setarray .shield_id[0], 2115, 2115;
	setarray .hgtop_id[0], rand(20,500), rand(20,500);
	setarray .hgmid_id[0], 125, 125;
	setarray .hglow_id[0], 0, 0;
	setarray .robe[0], 0, 0;
	setarray .options[0], 0, 0;
	setarray .clothcolor[0], rand(0,20), rand(0,20);
	setarray .mob_behavior[0], 1639, 1639;
	
	// Lord Knight
	setarray .lk_mob_names$, "Crocomom", "FrauD Mihk";
	setarray .lk_job_id[0], 4008, 4008;
	setarray .lk_gender[0], 1, 1;
	setarray .lk_hairstyle[0], rand(0,42), rand(0,42);
	setarray .lk_haircolor[0], rand(0,8), rand(0,8);
	setarray .lk_weapon_id[0], 1230, 1230;
	setarray .lk_shield_id[0], 2115, 2115;
	setarray .lk_hgtop_id[0], rand(20,500), rand(20,500);
	setarray .lk_hgmid_id[0], 125, 125;
	setarray .lk_hglow_id[0], 0, 0;
	setarray .lk_robe[0], 0, 0;
	setarray .lk_options[0], 0, 0;
	setarray .lk_clothcolor[0], rand(0,20), rand(0,20);
	setarray .lk_mob_behavior[0], 1634, 1634;
	
	setarray .ass_mob_names$, "Agendapice", "RatShanks";
	setarray .ass_job_id[0], 4013, 4013;
	setarray .ass_gender[0], 1, 0;
	setarray .ass_hairstyle[0], rand(0,42), rand(0,42);
	setarray .ass_haircolor[0], rand(0,8), rand(0,8);
	setarray .ass_weapon_id[0], 1257, 1257;
	setarray .ass_shield_id[0], 0, 0;
	setarray .ass_hgtop_id[0], rand(20,500), rand(20,500);
	setarray .ass_hgmid_id[0], 125, 125;
	setarray .ass_hglow_id[0], 0, 0;
	setarray .ass_robe[0], 0, 0;
	setarray .ass_options[0], 0, 0;
	setarray .ass_clothcolor[0], rand(0,20), rand(0,20);
	setarray .ass_mob_behavior[0], 1635, 1635;
	
	// Global variables are set when the script loads.
	$@EventMap$ = "poring_w01"; 
	set $@poringWinnerAID, 0; // Tracks the winner's Account ID. Initialized to 0 (no winner).
	set $@realPoringGID, 0; // Stores the Game ID (GID) of the unique "Real Poring". Initialized to 0.
	
	// This block runs on script load or reload.
	initnpctimer; // Initialize the NPC timer.
	setnpctimer 0; // Set timer to 0.
	startnpctimer; // Start the timer.
	disablenpc "PoringHuntWarper"; // Hide the warper NPC initially.
	killmonsterall $@EventMap$; // Clear any monsters on the event map at startup.
	end;

OnTimer5400000: // This label triggers every 90 minutes (5,400,000 milliseconds).
	// Main event trigger.
	setnpctimer 0; // Reset timer for the next cycle.
	startnpctimer; // Restart the timer.
	
	enablenpc "PoringHuntWarper"; // Make warper NPC visible for players to join.
	
	announce "A Poring Hunt event is about to begin!",bc_all; // Announce event start.
	sleep 1000; // Pause for 1 second.
	announce "Please go to @go 16 and click the Poring Hunt NPC to join Poring Hunt event!!",bc_all; // Instructions to join.

	// Countdown sequence before monster spawn.
	sleep 60000; // Pause for 1 minute.
	announce "Poring Hunt Event starting in 2 minutes!", bc_all; // 2-minute warning.
	sleep 60000; // Pause for 1 minute.
	announce "Poring Hunt Event starting in 1 minute!", bc_all; // 1-minute warning.
	sleep 1000; // Pause for 1 second.
	announce "Poring Hunt Event starting in 5 seconds!", bc_all; // 5-second countdown.
	sleep 1000; // Pause for 1 second.
	announce "Poring Hunt Event starting in 4 seconds!", bc_all; // 4-second countdown.
	sleep 1000; // Pause for 1 second.
	announce "Poring Hunt Event starting in 3 seconds!", bc_all; // 3-second countdown.
	sleep 1000; // Pause for 1 second.
	announce "Poring Hunt Event starting in 2 seconds!", bc_all; // 2-second countdown.
	sleep 1000; // Pause for 1 second.
	announce "Poring Hunt Event starting in 1 second!", bc_all; // 1-second countdown.
	sleep 1000; // Pause for 1 second.
	announce "Poring Hunt Event started! Lets Find the Real Poring!!", bc_all; // Event begins.
	
	disablenpc "PoringHuntWarper"; // Hide warper to prevent late joins.

	killmonsterall $@EventMap$; // Clear all existing monsters on the event map.
	
	// Spawn the fakeplayer
	for (.@i = 0; .@i < .num_fakePlayers; .@i++) {
		.hw_GID[.@i] = player_mbase($@EventMap$, 0, 0, .mob_names$[.@i], .job_id[.@i], .gender[.@i], .hairstyle[.@i], .haircolor[.@i], .weapon_id[.@i], .shield_id[.@i], .hgtop_id[.@i], .hgmid_id[.@i], .hglow_id[.@i], .robe[.@i], .options[.@i], .clothcolor[.@i], .mob_behavior[.@i], "");
		.lk_GID[.@i] = player_mbase($@EventMap$, 0, 0, .lk_mob_names$[.@i], .lk_job_id[.@i], .lk_gender[.@i], .lk_hairstyle[.@i], .lk_haircolor[.@i], .lk_weapon_id[.@i], .lk_shield_id[.@i], .lk_hgtop_id[.@i], .lk_hgmid_id[.@i], .lk_hglow_id[.@i], .lk_robe[.@i], .lk_options[.@i], .lk_clothcolor[.@i], .lk_mob_behavior[.@i], "");
		.ass_GID[.@i] = player_mbase($@EventMap$, 0, 0, .ass_mob_names$[.@i], .ass_job_id[.@i], .ass_gender[.@i], .ass_hairstyle[.@i], .ass_haircolor[.@i], .ass_weapon_id[.@i], .ass_shield_id[.@i], .ass_hgtop_id[.@i], .ass_hgmid_id[.@i], .ass_hglow_id[.@i], .ass_robe[.@i], .ass_options[.@i], .ass_clothcolor[.@i], .ass_mob_behavior[.@i], "");
	}
	// Spawn 100 "fake" Porings randomly.
	// monster $@EventMap$,0,0,"Poring",1002,100;
	
	areamonster $@EventMap$,88,117,120,90,"Poring",1002,100;

	// Spawn 1 "Real Poring" randomly, attaching a killer event.
	areamonster($@EventMap$, 88, 90, 120, 117, "Event Poring", 1002, 1, strnpcinfo(3) + "::OnRealPoringKilled");
	// set $@realPoringGID, monster($@EventMap$,0,0,"Event Poring",1002,1,strnpcinfo(3)+"::OnRealPoringKilled");
	
	end;

OnRealPoringKilled:

	if (playerattached()) {
		// Announce winner and reward.
		announce strcharinfo(0) + " has found and killed the Event Poring! Congratulations!", bc_all;

		getitem 30339, 5; // Give 5 Berry Tickets to the winner.
	} else if ($killermobgid > 0) {
		announce "The Event Poring was killed by [ " + getunitname($killermobgid) + "! ] Congratulations!", bc_all;
		$killermobgid = 0; // Reset the global temporary variable.
	}

	sleep 3000; // Pause for 3 seconds for messages to display.

	// Cleanup and warp players out.
	mapannounce $@EventMap$, "All remaining Porings have been vanquished!", bc_all; // Announce to event map.
	killmonsterall $@EventMap$; // Clear all remaining monsters on the map.
	
	
	// For testing
	// Safe cleanup loop that ignores GMs/Administrators
	.count = getmapunits(BL_PC, $@EventMap$, .players[0]);
	for (.i = 0; .i < .count; .i++) {
		if (attachrid(.players[.i])) {
			// Check if the player is a normal player (Group ID 0)
			if (getgroupid() == 0) {
				warp "prontera", 149, 161;
			}
			detachrid();
		}
	}
	
	// mapwarp $@EventMap$, "prontera", 149, 161; // Warp all players from event map to Prontera.
	end;

}

//===== rAthena Script =======================================
//= Poring Hunt Event Warper
//===== By: rAthena AI Assistant ============================
//= Warps players to the Poring Hunt event map.
//============================================================
pay_arche,53,133,4	script	PoringHuntWarper	1904,{ // NPC definition: Located at malaya (243, 216), facing south-east, using sprite ID 1904 (Bomb Poring).
	mes "[Poring Hunt Warper]"; // Display NPC's name in the dialog.
	mes "The Poring Hunt event is active! Would you like to go to the event map?"; // Prompt player to join.
	next; // Advance dialog.
	if (select("Yes","No") == 1) { // If player selects "Yes":
		warp "poring_w01",102,107; // Warp player to the event map at specific coordinates.
		end; // End script.
	}
	mes "[Poring Hunt Warper]"; // If player selects "No":
	mes "Alright, maybe next time!"; // Display farewell.
	close; // Close dialog and end script.
}
