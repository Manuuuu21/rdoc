//===== rAthena Script =======================================
//= Auto Asura NPC (Revised Combo Selection & Provokes)
//===== By: rAthena AI Assistant ============================
//= Automatically targets and casts Asura Strike combos.
//= Randomly chooses between DD and Normal combos per target.
//= Provokes the target upon its defeat.
//= @reloadnpcfile npc/custom/manuel_npc/autoasura.txt 
//============================================================
-	script	AutoAsuraNPC	-1,{
	OnInit:
		// Bind the commands
		bindatcmd "autoasura", strnpcinfo(3) + "::OnAutoAsura";
		bindatcmd "autoasuraoff", strnpcinfo(3) + "::OnAutoAsuraOff";
		end;

	OnAutoAsura:
		attachrid(getcharid(3));

		// Check if autoasura is already on
		if (.isAutoAsura[getcharid(3)]) {
			dispbottom "Autoasura is already On.";
			detachrid;
			end;
		}

		.isAutoAsura[getcharid(3)] = 1;
		@player_gid = getcharid(3); // This is the Account ID (AID)
		@target_gid = 0;
		@combo_type = rand(0, 1); // 0 = DD Asura, 1 = Normal Asura
		dispbottom "Autoasura On. Searching for players and mobs..." + " (Combo: " + (@combo_type ? "Normal" : "DD") + ")";

		// Start the main loop
		while (.isAutoAsura[getcharid(3)] && unitexists(@player_gid)) {
			attachrid(@player_gid);

			// --- YOUR REQUESTED BUFF CHECK ---
			if (!getstatus(SC_RAISINGDRAGON, 0, convertpcinfo(@player_gid, CPC_CHAR))) {
				unitstopwalk(@player_gid, USW_FORCE_STOP);
				.walking_status[getcharid(3)] = 0; 
				unitskilluseid @player_gid, "SR_RAISINGDRAGON", 10, @player_gid;
				sleep2 500; 
				continue;
			}
			// ---------------------------------

			// Get player's current location
			getmapxy(.@mapname$, .@x, .@y, BL_PC);

			// --- TARGET VALIDATION & LIFE CHECK ---
			.@target_type = getunittype(@target_gid);
			.@is_dead = 0;

			if (unitexists(@target_gid)) {
				if (.@target_type == BL_PC) {
					if (isdead(@target_gid)) .@is_dead = 1;
				} else if (.@target_type == BL_MOB) {
					if (getunitdata(@target_gid, .@mobdata) && .@mobdata[UMOB_HP] <= 0) .@is_dead = 1;
				}
			}

			// If target doesn't exist or is DEAD, change target
			if (!unitexists(@target_gid) || (.@target_type != BL_PC && .@target_type != BL_MOB) || .@is_dead) {
				
				// --- PROVOKING STATEMENT ON TARGET DEATH ---
				if (.@is_dead && unitexists(@player_gid)) {
					switch(rand(20)) {
						case 0: sleep2 2000; unittalk @player_gid, strcharinfo(0) + " : Is that all you got?"; break;
						case 1: sleep2 2000; unittalk @player_gid, strcharinfo(0) + " : Too easy!"; break;
						case 2: sleep2 2000; unittalk @player_gid, strcharinfo(0) + " : Next!"; break;
						case 3: sleep2 2000; unittalk @player_gid, strcharinfo(0) + " : Is there no one else?"; break;
						case 4: sleep2 2000; unittalk @player_gid, strcharinfo(0) + " : Pathetic."; break;
						case 5: sleep2 2000; unittalk @player_gid, strcharinfo(0) + " : Is this the best you can do?"; break;
						case 6: sleep2 2000; unittalk @player_gid, strcharinfo(0) + " : Who's next?"; break;
						case 7: sleep2 2000; unittalk @player_gid, strcharinfo(0) + " : Can't touch this!"; break;
						case 8: sleep2 2000; unittalk @player_gid, strcharinfo(0) + " : Piece of cake."; break;
						case 9: sleep2 2000; unittalk @player_gid, strcharinfo(0) + " : You need more practice."; break;
						case 10: sleep2 2000; unittalk @player_gid, strcharinfo(0) + " : Not even close!"; break;
						case 11: sleep2 2000; unittalk @player_gid, strcharinfo(0) + " : Bring on a real challenge!"; break;
						case 12: sleep2 2000; unittalk @player_gid, strcharinfo(0) + " : I'm just getting started."; break;
						case 13: sleep2 2000; unittalk @player_gid, strcharinfo(0) + " : Flawless victory."; break;
						case 14: sleep2 2000; unittalk @player_gid, strcharinfo(0) + " : Did you even try?"; break;
						case 15: sleep2 2000; unittalk @player_gid, strcharinfo(0) + " : Yawn... boring."; break;
						case 16: sleep2 2000; unittalk @player_gid, strcharinfo(0) + " : Too slow!"; break;
						case 17: sleep2 2000; unittalk @player_gid, strcharinfo(0) + " : Is that your full power?"; break;
						case 18: sleep2 2000; unittalk @player_gid, strcharinfo(0) + " : Better luck next time."; break;
						case 19: sleep2 2000; unittalk @player_gid, strcharinfo(0) + " : Down you go!"; break;
					}
				}
				// --------------------------------------------
	
				@target_gid = 0; 

				// Find nearby players and mobs within a 20x20 area
				.@nearby_count = getareaunits(BL_PC | BL_MOB, .@mapname$, .@x - 20, .@y - 20, .@x + 20, .@y + 20, .@nearby_units[0]);

				.@valid_count = 0;
				for (.@i = 0; .@i < .@nearby_count; .@i++) {
					.@potential_target = .@nearby_units[.@i];

					if (.@potential_target == @player_gid) continue;

					// Ignore dead targets while scanning
					.@p_type = getunittype(.@potential_target);
					if (.@p_type == BL_PC && isdead(.@potential_target)) continue;
					if (.@p_type == BL_MOB) {
						if (getunitdata(.@potential_target, .@p_mobdata) && .@p_mobdata[UMOB_HP] <= 0) continue;
					}

					set .@valid_targets[.@valid_count], .@potential_target;
					.@valid_count++;
				}

				if (.@valid_count > 0) {
					@target_gid = .@valid_targets[0];
					
					dispbottom "Target found: " + getunitname(@target_gid) + " (Combo: " + (@combo_type ? "Normal" : "DD") + ")";
				} else {
					dispbottom "No valid targets found, searching...";
					sleep2 500; 
					continue; 
				}
			}

			// Get current positions
			getmapxy(.@map$, .@px, .@py, BL_PC);
			getmapxy(.@tmap$, .@tx, .@ty, getunittype(@target_gid), @target_gid);

			if (.@map$ != .@tmap$) {
				@target_gid = 0;
				sleep2 50; 
				continue; 
			}

			// Check distance to target
			.@distance = distance(.@px, .@py, .@tx, .@ty);

			// If target is within Asura range (<= 2 cells)
			if (.@distance <= 2) {
				if (.walking_status[getcharid(3)] == 1) {
					unitstopwalk(@player_gid, USW_FORCE_STOP);
					.walking_status[getcharid(3)] = 0; 
				}

				// Perform Asura combo
				if (unitexists(@player_gid) && unitexists(@target_gid)) {
					if (@combo_type == 0) {
						// DD Asura Combo
						percentheal 100, 100;
						unitskilluseid @player_gid, 401, 1, @player_gid; // Dangerous Soul Collect
						sleep2 100;
						unitskilluseid @player_gid, 270, 5, @player_gid; // Explosion Spirits
						percentheal 100, 100;
						sleep2 300;
						unitskilluseid @player_gid, 271, 5, @target_gid; // Asura Strike
						percentheal 100, 100;
					} else {
						// Normal Combo Asura
						percentheal 100, 100;
						unitskilluseid @player_gid, 401, 1, @player_gid; // Dangerous Soul Collect
						sleep2 100;
						unitskilluseid @player_gid, 270, 5, @player_gid; // Explosion Spirits
						percentheal 100, 100;
						sleep2 100;
						unitskilluseid @player_gid, 401, 1, @player_gid; // Dangerous Soul Collect
						percentheal 100, 100;
						sleep2 300;
						unitskilluseid @player_gid, 271, 5, @target_gid; // Asura Strike
						percentheal 100, 100;
					}
				}

				sleep2 20; 
				continue; 
			} else {
				// Target is outside range, walk to it
				if (unitexists(@player_gid)) {
					unitwalkto(@player_gid, @target_gid);
					.walking_status[getcharid(3)] = 1;
				}

				sleep2 150; 
				continue; 
			}
		}

		dispbottom "Autoasura is Off.";
		if (unitexists(@player_gid)) {
			unitstopwalk(@player_gid, USW_FORCE_STOP);
			.walking_status[getcharid(3)] = 0;
		}
		detachrid;
		end;

	OnAutoAsuraOff:
		attachrid(getcharid(3));

		if (.isAutoAsura[getcharid(3)]) {
			.isAutoAsura[getcharid(3)] = 0;
		} else {
			dispbottom "Autoasura is already Off.";
		}

		detachrid; 
		end;
}
