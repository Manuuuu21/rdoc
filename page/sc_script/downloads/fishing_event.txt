// Shared catch table for clicking the NPC and @autofish. Zero means no catch.
function	script	F_FishingCatchItem	{
	.@roll = getarg(0);
	if (.@roll < 20) return 6771;
	if (.@roll < 35) return 6772;
	if (.@roll < 50) return 6773;
	if (.@roll < 60) return 6775;
	if (.@roll < 70) return 6778;
	if (.@roll < 75) return 6770;
	if (.@roll < 85) return 30338;
	return 0;
}

function	script	F_FishingSpotInRange	{
	if (getarg(3) == "") return 0;
	if (getmapxy(.@map$, .@x, .@y, BL_NPC, getarg(3))) return 0;
	return getarg(0) == .@map$ && distance(getarg(1), getarg(2), .@x, .@y) <= getarg(4);
}

ayothaya,261,95,3	script	Fishing Spot	10065,{
	if (@AutoFishing) {
		dispbottom "Autofishing is active. Use @autofish off to stop.";
		end;
	}
	// Fishing Event item name
	// 1. Fishing Rod (30006)
	// 2. Fishing Bots (2443)
	// 3. Fishing Muffler (2550)
	// 4. Fishing Hat (30005)
	// 5. Artificial Lure (30340)
	setarray .fishingItems_itemID, 30006, 2443, 2550, 30005, 30340;

	// For itemID caught fish.
	// 1. Tuna (6771)
	// 2. Octopus (6772)
	// 3. Snapper (6773)
	// 4. Salmon (6775)
	// 5. Squid (6778)
	// 6. Shark (6770)
	// 7. Marlin (30338)
	setarray .caughtFishItemID, 6771, 6772, 6773, 6775, 6778, 6770, 30338;

	// Get the character ID of the player who is fishing
	.@char_id = getcharid(3);

	// Script to check if the player has the Fishing Rod weapon equipped
	// and has an artificial lure
	if (isequipped(.fishingItems_itemID[0])) {
		if (countitem(.fishingItems_itemID[4]) == 0) {
			mes "You do not have a ^1100AAArtificial Lure^000000.";
	    	close;
		}
		else {
			if (isFishing[.@char_id]) {
				// Show some messege above the head of character
				showscript "You're currently fishing! Please wait until it's done.", .@char_id, self;
				end;
			}
			else {
				// Casting time for Fishing
				set .@fcast,9;

				if (isequipped(.fishingItems_itemID[0])) { //Fishing Rod
					set .@fcast,.@fcast - 3;
				}
				if (isequipped(.fishingItems_itemID[1])) { //Fishing Boots
					set .@fcast,.@fcast - 2;
				}
				if (isequipped(.fishingItems_itemID[2])) { //Fishing Muffler
					set .@fcast,.@fcast - 2;
				}
				if (isequipped(.fishingItems_itemID[3])) { //Fishing Hat
					set .@fcast,.@fcast - 1;
				}

				// Show a message above the character's head indicating the action
				showscript "Throwing lure!", .@char_id, self;

				// Display a progress bar to indicate casting time
				progressbar "ffffff", .@fcast;

				// Prevents the current char from moving away.
				pcblockmove .@char_id,1;

				// Set the flag to indicate that this player is currently fishing
				set isFishing[.@char_id], 1;

				// Show some messege above the head of character
				showscript "You're fishing!", .@char_id, self;

				// Trigger a bubble special effect at the fishing spot
				// and Play a water splash sound effect
				specialeffect EF_BUBBLE, AREA, strnpcinfo(3);
				soundeffect "splash_water.wav", 0;

				// Display a fishing cutin image (image must be in the correct directory)
				cutin "2013_summer_fish_1_edit", 1;
				// Wait for a short moment for effect
				sleep2 500;
				cutin "2013_summer_fish_1_edit", 255;
				cutin "2013_summer_fish_2_edit", 1;
				sleep2 800;
				cutin "2013_summer_fish_2_edit", 255;
				cutin "2013_summer_fish_3_edit", 1;
				sleep2 800;
				cutin "2013_summer_fish_3_edit", 255;
				cutin "2013_summer_fish_4_edit", 1;
				sleep2 800;
				cutin "2013_summer_fish_4_edit", 255;

				.@fish = callfunc("F_FishingCatchItem", rand(100));
				if (.@fish) {
					cutin "2013_summer_fish_5_edit", 1;
					getitem .@fish, 1;
				} else {
					specialeffect2 611;
					cutin "2013_summer_fish_6_edit", 1; // No Catched 
					// You didn't catch anything this time. 15% chance
				}

				// Play a sound effect for players when a fish is caught
				soundeffect "catched_fish.wav", 0;
				
				// Remove 1 piece of lure from the player's inventory after fishing
				delitem .fishingItems_itemID[4], 1;

				sleep2 800;
				cutin "", 255;
				goto L_Fishing_Complete;
			}
		}
	} else {
	    mes "You do not have the ^1100AASmall Fishing Rod^000000 weapon equipped.";
	    close;
	}

	// Fishing completion logic after sleep
	L_Fishing_Complete:
		// Reset the flag after fishing is done
		set isFishing[.@char_id], 0;

		// Show some messege above the head of character
		showscript "You finished fishing!", .@char_id, self;

		// Enables the current char to move again.
		pcblockmove .@char_id,0;
	end;
}

// Player timers separate the casting bar and fishing animations.
-	script	AutoFishing	-1,{
OnInit:
	.Range = 10; // Maximum distance; standing closer is allowed.
	.Rod = 30006;
	.Lure = 30340;
	setarray .Spots$[0], "Fishing Spot", "Fishing Spot#fs2", "Fishing Spot#fs3",
		"Fishing Spot#fs4", "Fishing Spot#fs5", "Fishing Spot#fs6", "Fishing Spot#fs7",
		"Fishing Spot#fs8", "Fishing Spot#fs9", "Fishing Spot#fs10", "Fishing Spot#fs11",
		"Fishing Spot#fs12", "Fishing Spot#fs13", "Fishing Spot#fs14";
	bindatcmd "autofish", strnpcinfo(3) + "::OnCommand";
	end;

OnCommand:
	// Bound commands queued during a progress bar lose their parameters in
	// this emulator. Treat a queued command as stop, never as another start.
	if (.@atcmd_command$ == "") {
		if (@AutoFishing) callsub S_Stop, "Autofishing stopped.";
		else dispbottom "Use @autofish after closing your current NPC interaction.";
		end;
	}
	if (.@atcmd_numparameters == 1 && strtolower(.@atcmd_parameters$[0]) == "off") {
		if (!@AutoFishing) {
			dispbottom "Autofishing is already off.";
			end;
		}
		callsub S_Stop, "Autofishing stopped.";
		end;
	}
	if (.@atcmd_numparameters) {
		dispbottom "Usage: @autofish or @autofish off";
		end;
	}
	if (@AutoFishing) {
		dispbottom "Autofishing is already active. Use @autofish off to stop.";
		end;
	}
	if (isFishing[getcharid(3)]) {
		dispbottom "Please finish your current fishing attempt first.";
		end;
	}
	@AutoFishSpot$ = "";
	getmapxy .@map$, .@x, .@y, BL_PC;
	.@nearest = .Range + 1;
	for (.@i = 0; .@i < getarraysize(.Spots$); .@i++) {
		if (getmapxy(.@npcmap$, .@nx, .@ny, BL_NPC, .Spots$[.@i])) continue;
		if (.@npcmap$ != .@map$) continue;
		.@distance = distance(.@x, .@y, .@nx, .@ny);
		if (.@distance < .@nearest) {
			@AutoFishSpot$ = .Spots$[.@i];
			.@nearest = .@distance;
		}
	}
	.@reason$ = callsub(S_Check);
	if (.@reason$ != "") {
		dispbottom .@reason$;
		end;
	}
	@AutoFishing = 1;
	@AutoFishPhase = 0;
	pcblockmove getcharid(3), 1;
	@AutoFishLocked = 1;
	deltimer strnpcinfo(3) + "::OnStep";
	dispbottom "Autofishing started. Use @autofish off to stop.";
	addtimer 1, strnpcinfo(3) + "::OnStep";
	end;

OnStep:
	if (!@AutoFishing) end;
	.@reason$ = callsub(S_Check);
	if (.@reason$ != "") {
		callsub S_Stop, .@reason$;
		end;
	}
	switch (@AutoFishPhase) {
	case 0:
		face_npc @AutoFishSpot$;
		// Same cast-time bonuses as manual fishing: 6 seconds down to 1.
		.@fcast = 6 - 2 * isequipped(2443) - 2 * isequipped(2550) - isequipped(30005);
		isFishing[getcharid(3)] = 1;
		showscript "Throwing lure!", getcharid(3), self;
		@AutoFishCasting = 1;
		// If the client cancels the bar, the script ends before its next line.
		// The watchdog then releases the movement lock and cancels autofishing.
		addtimer .@fcast * 1000 + 1000, strnpcinfo(3) + "::OnCastTimeout";
		progressbar "ffffff", .@fcast;
		@AutoFishCasting = 0;
		deltimer strnpcinfo(3) + "::OnCastTimeout";
		if (!@AutoFishing) end;
		.@delay = 1; // The progress bar already waited for the casting time.
		break;
	case 1:
		showscript "You're fishing!", getcharid(3), self;
		specialeffect EF_BUBBLE, AREA, @AutoFishSpot$;
		soundeffect "splash_water.wav", 0;
		cutin "2013_summer_fish_1_edit", 1;
		.@delay = 500;
		break;
	case 2:
	case 3:
	case 4:
		cutin "2013_summer_fish_" + @AutoFishPhase + "_edit", 1;
		.@delay = 800;
		break;
	case 5:
		.@fish = callfunc("F_FishingCatchItem", rand(100));
		// Script && evaluates both operands; never pass the no-catch ID to checkweight.
		if (.@fish) {
			if (!checkweight(.@fish, 1)) {
				callsub S_Stop, "Autofishing stopped: make room in your inventory.";
				end;
			}
		}
		// Charge once, including unsuccessful attempts, before granting a catch.
		delitem .Lure, 1;
		if (.@fish) {
			getitem .@fish, 1;
			cutin "2013_summer_fish_5_edit", 1;
			// dispbottom "Caught " + getitemname(.@fish) + "!";
		} else {
			specialeffect2 611;
			cutin "2013_summer_fish_6_edit", 1;
		}
		soundeffect "catched_fish.wav", 0;
		if (!countitem(.Lure)) {
			callsub S_Stop, "Autofishing stopped: no Artificial Lure remaining.";
			end;
		}
		.@delay = 800;
		break;
	case 6:
		cutin "", 255;
		@AutoFishPhase = 0;
		addtimer 500, strnpcinfo(3) + "::OnStep";
		end;
	}
	@AutoFishPhase++;
	addtimer .@delay, strnpcinfo(3) + "::OnStep";
	end;

OnCastTimeout:
	if (@AutoFishing && @AutoFishCasting)
		callsub S_Stop, "Autofishing stopped: casting was interrupted.";
	end;

S_InRange:
	if (getmapxy(.@map$, .@x, .@y, BL_PC)) return 0;
	return callfunc("F_FishingSpotInRange", .@map$, .@x, .@y, getarg(0), .Range);

S_Check:
	if (isdead()) return "Autofishing stopped: you are dead.";
	if (!callsub(S_InRange, @AutoFishSpot$))
		return "Autofishing requires a Fishing Spot within " + .Range + " cells.";
	if (!isequipped(.Rod)) return "Autofishing stopped: equip your Small Fishing Rod.";
	if (!countitem(.Lure)) return "Autofishing stopped: no Artificial Lure remaining.";
	return "";

S_Stop:
	deltimer strnpcinfo(3) + "::OnStep";
	deltimer strnpcinfo(3) + "::OnCastTimeout";
	if (@AutoFishLocked) pcblockmove getcharid(3), 0;
	cutin "", 255;
	isFishing[getcharid(3)] = 0;
	@AutoFishing = 0;
	@AutoFishLocked = 0;
	@AutoFishPhase = 0;
	@AutoFishCasting = 0;
	@AutoFishSpot$ = "";
	if (getarg(0, "") != "") dispbottom getarg(0);
	return;

OnPCDieEvent:
OnPCLogoutEvent:
	if (@AutoFishing) callsub S_Stop;
	end;

OnPCLoginEvent:
	// Character timers never survive logout; clear any saved fishing flag too.
	isFishing[getcharid(3)] = 0;
	end;
}

-	script	joinFishing	-,{
	
	OnInit:
		bindatcmd "joinfishing", strnpcinfo(3) + "::OnFishing";
	end;

	OnFishing:
		// Attach the player who used the command
		attachrid(getcharid(3));

		// Define coordinates and maps
		setarray .@hugelCoordinates[0], 205, 208;
		setarray .@ayothayaCoordinates[0], 231, 100;
		setarray .@map$, "ayothaya", "hugel";

		// Randomly choose Ayothaya or Hugel
		.@choice = rand(2);
		
		// Set coordinates based on choice
		if (.@choice == 0) {
			.@map$ = .@map$[0];
			.@x = .@ayothayaCoordinates[0];
			.@y = .@ayothayaCoordinates[1];
		} else {
			.@map$ = .@map$[1];
			.@x = .@hugelCoordinates[0];
			.@y = .@hugelCoordinates[1];
		}

		// Warp the player
		warp .@map$, .@x, .@y;
	end;
}
