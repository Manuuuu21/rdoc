//===== rAthena Script =======================================
//= Clone Spawner
//===== By: rAthena AI Assistant ============================
//= Spawns a clone of the player who clicks the NPC.
//= The clone will automatically respawn a few seconds after being killed.
//= The clone will now autotarget the cloned player with a skill.
//=	@reloadnpcfile npc/custom/manuel_npc/clone_master.txt
//============================================================
-	script	cloneCmd	-,{

	// This block runs once when the NPC script is loaded (e.g., during server startup or @reloadscript).
	OnInit:
		// Binds the "@msclone" GM command to the "OnBokClone" label within this NPC.
		// - "msclone": The command name.
		// - strnpcinfo(0) + "::OnBokClone": The event label to trigger (this NPC's unique name followed by the label).
		// - 0: Minimum GM level for the command (player level 0).
		// - 99: Maximum GM level for the command (GM level 99).
		bindatcmd("msclone", strnpcinfo(0) + "::OnBokClone", 0, 99);
		end;
	
	// This label is triggered when the "@msclone" command is used by a GM.
	OnBokClone:
		attachrid(getcharid(0));
		// Retrieves parameters passed with the GM command.
		// @atcmd_parameters$[0]: First parameter (expected: target character's Account ID / GID).
		// @atcmd_parameters$[1]: Second parameter (expected: map name for spawning).
		// @atcmd_parameters$[2]: Third parameter (expected: number of clones to spawn).
		set $@charid, atoi(.@atcmd_parameters$[0]); // Stores the target player's Account ID (RID) in a global temporary variable.
		set $@map$, .@atcmd_parameters$[1];          // Stores the target map name in a global temporary variable.
		set .@count, atoi(.@atcmd_parameters$[2]);   // Stores the number of clones to spawn.

		// Input validation: Checks if required parameters are valid.
		if ($@charid <= 0 || $@map$ == "" || .@count <= 0) {
			dispbottom "Usage: @msclone <char_id> <map> <count>"; // Informs the GM of incorrect usage.
			end; // Terminates the script if input is invalid.
		}
		
		// set the top headgear
		setarray .itemHGTop, 99, 513, 1301;
		// Set the low headgear
		setarray .itemHGBottom, 2204, 2205, 2206, 2209;
		// set the job Class
		setarray .jobclassTarget, 4016, 4077, 4262;
		// set RobeID
		setarray .itemRobe, 23, 25, 1, 12, 14, 16;
		
		// Pre-calculate the combined mob mode flags to avoid "player not attached!" errors.
		set $@mob_mode_flags, MD_AGGRESSIVE | MD_SKILL | MD_ANGRY | MD_LOOTER;
		
		// Loop to spawn the specified number of clones.
		for (.@i = 0; .@i < .@count; .@i++) {
			
			// Spawns a clone of the target player.
			// - $@map$, .@x, .@y: Map and coordinates for the clone's spawn.
			// - strnpcinfo(0)+"::OnMSCloneKilled": The event label to call when this specific clone is killed.
			//   This ensures that each clone's death triggers a respawn.
			// - $@charid: The Account ID (GID) of the player to clone.
			// - 0: Master ID (0 means no master).
			// - 0: Mob mode (default).
			// - 1: Flag for summoned monster (targets other monsters).
			set .@cloneGID, clone($@map$, 0, 0, strnpcinfo(0)+"::OnMSCloneKilled", $@charid, 0, 0, 1);
			
			setunitdata .@cloneGID, UMOB_MAXHP, 990000;
			setunitdata .@cloneGID, UMOB_HP, 990000;
			setunitdata .@cloneGID, UMOB_STR, 1000;
			setunitdata .@cloneGID, UMOB_AGI, 10000;
			setunitdata .@cloneGID, UMOB_DEX, 10000;
			setunitdata .@cloneGID, UMOB_LUK, 1000;
			// Randomly Choose RobeID
			setunitdata .@cloneGID, UMOB_ROBE, .itemRobe[rand(getarraysize(.itemRobe))];
			// Set the clone's gender (0 for female, 1 for male).
			setunitdata .@cloneGID, UMOB_SEX, rand(0, 1);
			// Set Top, Bottom Headgear and Job Class
			setunitdata .@cloneGID, UMOB_HEADTOP, .itemHGTop[rand(getarraysize(.itemHGTop))];
			setunitdata .@cloneGID, UMOB_HEADBOTTOM, .itemHGBottom[rand(getarraysize(.itemHGBottom))];
			setunitdata .@cloneGID, UMOB_CLASS, .jobclassTarget[rand(getarraysize(.jobclassTarget))];
		}
		
		// Set the mobai into attack
		setunitdata .@cloneGID, UMOB_AI, AI_ATTACK;
		// Make the clone aggressive, able to move, and use skills.
		setunitdata .@cloneGID, UMOB_MODE, MD_AGGRESSIVE | MD_SKILL | MD_ANGRY | MD_LOOTER; // Use the pre-calculated flag.

		// Announces to the map that clones have been spawned.
		// strcharinfo(3) gets the name of the map where the command was issued.
		mapannounce strcharinfo(3), "Spawned clones of ["+$@charid+"] in map ["+$@map$+"]", bc_blue;
		end; // Ends the OnBokClone script instance.

	// This label is triggered when a cloned monster is killed.
	OnMSCloneKilled:
		
		set .@player_aid, playerattached();
		// set the top headgear
		setarray .itemHGTop, 99, 513, 1301;
		// Set the low headgear
		setarray .itemHGBottom, 2204, 2205, 2206, 2209;
		// set the job Class
		setarray .jobclassTarget, 4016, 4077, 4262;
		// set RobeID
		setarray .itemRobe, 23, 25, 1, 12, 14, 16;

		if (.@player_aid != 0) {
			// This creates a continuous respawn cycle.
			// The GID of the *newly spawned* clone for this specific event instance is stored in .@cloneGID.
			set .@cloneGID, clone($@map$, 0, 0, strnpcinfo(0)+"::OnMSCloneKilled", $@charid, 0, 0, 1);
			announce "Killed by Player: Respawned clones of ["+$@charid+"] in map ["+$@map$+"]", bc_blue;
			
			setunitdata .@cloneGID, UMOB_MAXHP, 990000;
			setunitdata .@cloneGID, UMOB_HP, 990000;
			setunitdata .@cloneGID, UMOB_STR, 1000;
			setunitdata .@cloneGID, UMOB_AGI, 10000;
			setunitdata .@cloneGID, UMOB_DEX, 10000;
			setunitdata .@cloneGID, UMOB_LUK, 1000;
			// Randomly Choose RobeID
			setunitdata .@cloneGID, UMOB_ROBE, .itemRobe[rand(getarraysize(.itemRobe))];
			// Set the clone's gender (0 for female, 1 for male).
			setunitdata .@cloneGID, UMOB_SEX, rand(0, 1);
			// Set Top, Bottom Headgear and Job Class
			setunitdata .@cloneGID, UMOB_HEADTOP, .itemHGTop[rand(getarraysize(.itemHGTop))];
			setunitdata .@cloneGID, UMOB_HEADBOTTOM, .itemHGBottom[rand(getarraysize(.itemHGBottom))];
			setunitdata .@cloneGID, UMOB_CLASS, .jobclassTarget[rand(getarraysize(.jobclassTarget))];
			
			// Set the mobai into attack
			setunitdata .@cloneGID, UMOB_AI, AI_ATTACK;
			// Make the clone aggressive, able to move, and use skills.
			setunitdata .@cloneGID, UMOB_MODE, MD_AGGRESSIVE | MD_SKILL | MD_ANGRY | MD_LOOTER; // Use the pre-calculated flag.
		}
		else {
			// This creates a continuous respawn cycle.
			// The GID of the *newly spawned* clone for this specific event instance is stored in .@cloneGID.
			set .@cloneGID, clone($@map$, 0, 0, strnpcinfo(0)+"::OnMSCloneKilled", $@charid, 0, 0, 1);
			announce "Killed by Mobs: Respawned clones of ["+$@charid+"] in map ["+$@map$+"]", bc_blue;
			
			setunitdata .@cloneGID, UMOB_MAXHP, 990000;
			setunitdata .@cloneGID, UMOB_HP, 990000;
			setunitdata .@cloneGID, UMOB_STR, 1000;
			setunitdata .@cloneGID, UMOB_AGI, 10000;
			setunitdata .@cloneGID, UMOB_DEX, 10000;
			setunitdata .@cloneGID, UMOB_LUK, 1000;
			// Randomly Choose RobeID
			setunitdata .@cloneGID, UMOB_ROBE, .itemRobe[rand(getarraysize(.itemRobe))];
			// Set the clone's gender (0 for female, 1 for male).
			setunitdata .@cloneGID, UMOB_SEX, rand(0, 1);
			// Set Top, Bottom Headgear and Job Class
			setunitdata .@cloneGID, UMOB_HEADTOP, .itemHGTop[rand(getarraysize(.itemHGTop))];
			setunitdata .@cloneGID, UMOB_HEADBOTTOM, .itemHGBottom[rand(getarraysize(.itemHGBottom))];
			setunitdata .@cloneGID, UMOB_CLASS, .jobclassTarget[rand(getarraysize(.jobclassTarget))];
			
			// Set the mobai into attack
			setunitdata .@cloneGID, UMOB_AI, AI_ATTACK;
			// Make the clone aggressive, able to move, and use skills.
			setunitdata .@cloneGID, UMOB_MODE, $@mob_mode_flags; // Use the pre-calculated flag.
		}
		end; 
		
}
