//===== rAthena Script =======================================
//= DailyAnnouncer
//===== By: rAthena AI Assistant ============================
//= Announces daily events and various holidays at specific times.
//============================================================
-	script	HolidayAnnouncer	-1,{
OnDay0101:
	// This event triggers every year on January 1st.
	announce "GM-Bokiebok: Happy New Year, adventurers! May your year be filled with epic quests and bountiful loot!",bc_all;
	end;

OnClock0000:
	// This event triggers every day at 00:00 (12:00 AM / Midnight).
	// We need to check if it's specifically December 31st.
	if (gettime(DT_MONTH) == DECEMBER && gettime(DT_DAYOFMONTH) == 31) {
		announce "GM-Bokiebok: Countdown to New Year! It's officially New Year's Eve!!",bc_all;
	}
	end;

OnDay0214:
	// This event triggers every year on February 14th (Valentine's Day).
	announce "GM-Bokiebok: Happy Valentine's Day! Share the love (and loot!) with your fellow adventurers!",bc_all;
	end;

OnDay0317:
	// This event triggers every year on March 17th (St. Patrick's Day).
	announce "GM-Bokiebok: Happy St. Patrick's Day! May the luck of the Irish be with you on your adventures!",bc_all;
	end;

OnDay1101:
	// This event triggers every year on October 31st (Halloween).
	announce "GM-Bokiebok: Happy Halloween! Beware of spooky monsters and trick-or-treat for rare items!",bc_all;
	end;

OnDay1225:
	// This event triggers every year on December 25th (Christmas).
	announce "GM-Bokiebok: Merry Christmas, everyone! Wishing you a joyful day filled with gifts and good cheer!",bc_all;
	end;
}
