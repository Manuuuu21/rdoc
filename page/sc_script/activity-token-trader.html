      Activity Token Trader

      A trader opens upper, middle, and lower costume-headgear shops that use Activity Tokens as currency.

      activity_token_trader.txt
      //===== By:==================================================
//= Manuel Sintos
// @reloadnpc npc/custom/manuel_npc/activity_token_trader.txt

prontera,164,158,3	script	Activity Token Trader	564,{

	switch(select("Costume Headgear (Upper):Costume Headgear (Middle):Costume Headgear (Lower)")) {
		case 1:
			callshop "cHGupper",0;
			end;
		break;
		case 2:
			callshop "cHGmiddle",0;
			end;
		break;
		case 3:
			callshop "cHGlower",0;
			end;
		break;
	}

}
